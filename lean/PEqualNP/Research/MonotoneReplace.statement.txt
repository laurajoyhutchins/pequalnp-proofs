∀ (P N : ℕ → Prop) (c : ℕ → Bool → Bool) (u v : ℕ → Bool),
    (∀ x, c x false = true → c x true = true) →
    (∀ x, P x → u x = true → v x = true) →
    (∀ x, N x → v x = true → u x = true) →
    (∀ x, P x → c x (u x) = true) →
    (∀ x, N x → c x (u x) = false) →
    (∀ x, P x → c x (v x) = true) ∧
    (∀ x, N x → c x (v x) = false)
